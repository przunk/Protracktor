# UADE -- GPL (UAE, no version stated) and LGPL (libuade); `docs/LICENSES.md` has the reading. The Amiga custom formats: TFMX, Hippel, David Whittaker, SidMon,
# Sonic Arranger, Delta Music and a hundred more. It emulates an Amiga and runs the original replay
# routines, which is why it reaches music no tracker library can.
#
# **This is the one backend that is two programs** (`docs/BACKLOG.md` A44, decided 2026-09-19).
# libuade does not decode anything: it opens a socketpair, forks, and execs `uadecore`, which is the
# emulator. The alternative -- running `uadecore_main` on a thread inside the app, where the seam is
# one function -- founders on 51 `exit()` calls in the command loop, forty of them in `uade.c`. One
# of those fires when the *emulated Amiga program* asks for a file that is not there, which is
# ordinary damaged data (`docs/STATUS.md` C42, C55). `exit()` is not an exception: the guard on the
# engine boundary catches nothing and the app disappears. Patching 51 exits is a fork of a library
# §3 of docs/ARCHITECTURE.md says we do not fork.
#
# So upstream's process model is kept, and the emulator ships as `lib/<abi>/libuadecore.so` -- an
# executable under a library's name, because `nativeLibraryDir` is the one place Android still
# permits executing from. `docs/PLAN_FORMATS.md` §4 has the measurement and the trade, including
# what it costs: the browser cannot have UADE at all, since `fork` and `exec` do not exist in
# WebAssembly.
#
# Four targets, and the reason they are four:
#
#   uade_zakalwe    the author's support library; both halves need it
#   uade_bencode    bencode, for RMC containers; only libuade reads those
#   uade_frontend   libuade -- identification, configuration, IPC. Links into the engine.
#   uadecore        the emulator. Its own process, its own binary, nothing shared but the socket.
set(UADE "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/uade")
set(ZAKALWE "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/libzakalwe")
set(BENCODE "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/bencodetools")

# `cpuemu.c` is generated, not cloned, so this checks for it rather than for a file git would have
# brought: a tree fetched by an older version of the script would otherwise fail three hundred
# undefined symbols later, with nothing pointing back here.
if(NOT EXISTS "${UADE}/src/cpuemu.c")
    message(FATAL_ERROR
        "UADE sources are missing or incomplete. Run ./scripts/fetch-uade.py first.\n"
        "  expected: ${UADE}/src/cpuemu.c (generated from table68k by the host compiler)")
endif()

# --- libzakalwe -----------------------------------------------------------------------------
# The modules upstream's own Makefile puts in the library, minus the ones it lists separately as
# "next" (digraph, hash64, thread): nothing here reaches them.
add_library(uade_zakalwe STATIC
    "${ZAKALWE}/array.c" "${ZAKALWE}/base.c" "${ZAKALWE}/bitfield.c" "${ZAKALWE}/endianess.c"
    "${ZAKALWE}/file.c" "${ZAKALWE}/input.c" "${ZAKALWE}/math.c" "${ZAKALWE}/mem.c"
    "${ZAKALWE}/norm_inv_table_65536.c" "${ZAKALWE}/random.c" "${ZAKALWE}/str_array.c"
    "${ZAKALWE}/string.c" "${ZAKALWE}/time.c" "${ZAKALWE}/unix-support.c")
target_include_directories(uade_zakalwe PUBLIC "${ZAKALWE}/include")
target_compile_options(uade_zakalwe PRIVATE -w)
set_target_properties(uade_zakalwe PROPERTIES POSITION_INDEPENDENT_CODE ON)

# --- bencodetools ---------------------------------------------------------------------------
# One file, and its header sits under `include/` in the shape the sources ask for it:
# `#include <bencodetools/bencode.h>`.
add_library(uade_bencode STATIC "${BENCODE}/bencode.c")
target_include_directories(uade_bencode PUBLIC "${BENCODE}/include")
target_compile_options(uade_bencode PRIVATE -w)
set_target_properties(uade_bencode PROPERTIES POSITION_INDEPENDENT_CODE ON)

# --- libuade --------------------------------------------------------------------------------
# COMMONMODULES from src/frontends/common/Makefile.common, in upstream's order. `ossupport.c` is
# generated by the fetch script the way configure generates it -- it is `unixsupport.c` plus the
# replacement for `canonicalize_file_name`, which glibc has and bionic does not.
set(UADE_COMMON "${UADE}/src/frontends/common")
add_library(uade_frontend STATIC
    "${UADE_COMMON}/unixatomic.c" "${UADE_COMMON}/uadeipc.c" "${UADE_COMMON}/amifilemagic.c"
    "${UADE_COMMON}/eagleplayer.c" "${UADE_COMMON}/unixwalkdir.c" "${UADE_COMMON}/effects.c"
    "${UADE_COMMON}/uadecontrol.c" "${UADE_COMMON}/uadeconf.c" "${UADE_COMMON}/uadestate.c"
    "${UADE_COMMON}/uadeutils.c" "${UADE_COMMON}/md5.c" "${UADE_COMMON}/ossupport.c"
    "${UADE_COMMON}/rmc.c" "${UADE_COMMON}/songdb.c" "${UADE_COMMON}/songinfo.c"
    "${UADE_COMMON}/vparray.c" "${UADE_COMMON}/support.c" "${UADE_COMMON}/fifo.c")
target_include_directories(uade_frontend
    PUBLIC "${UADE}/src/frontends/include"
    PRIVATE "${UADE_COMMON}")
target_link_libraries(uade_frontend PUBLIC uade_zakalwe uade_bencode)
target_compile_options(uade_frontend PRIVATE -w)
set_target_properties(uade_frontend PROPERTIES POSITION_INDEPENDENT_CODE ON)

# --- uadecore -------------------------------------------------------------------------------
# **The 68000 emulator is compiled eight times.** `cpuemu.c` is 80,000 lines of generated switch
# statement and upstream splits it into eight translation units with -DPART_1..8, because compiling
# it whole takes minutes and more memory than some machines have. One file per part is the same
# split, written where CMake can see it: a target cannot give two sets of defines to one source.
set(CPUEMU_PARTS "")
foreach(part RANGE 1 8)
    set(wrapper "${CMAKE_CURRENT_BINARY_DIR}/cpuemu${part}.c")
    file(WRITE "${wrapper}" "#define PART_${part}\n#include \"cpuemu.c\"\n")
    list(APPEND CPUEMU_PARTS "${wrapper}")
endforeach()

# Named `libuadecore.so` although it is an executable, and that is the whole trick: Android extracts
# and keeps `lib/<abi>/lib*.so`, and `nativeLibraryDir` is mounted executable. Anywhere else in an
# app's storage is not, since Android 10.
add_executable(uadecore
    "${UADE}/src/main.c" "${UADE}/src/newcpu.c" "${UADE}/src/memory.c" "${UADE}/src/custom.c"
    "${UADE}/src/cia.c" "${UADE}/src/audio.c" "${UADE}/src/cpustbl.c" "${UADE}/src/missing.c"
    "${UADE}/src/sd-sound.c" "${UADE}/src/md-support.c" "${UADE}/src/cfgfile.c"
    "${UADE}/src/fpp.c" "${UADE}/src/debug.c" "${UADE}/src/readcpu.c" "${UADE}/src/cpudefs.c"
    ${CPUEMU_PARTS}
    "${UADE}/src/uade.c" "${UADE}/src/uademain.c" "${UADE}/src/sinctable.c"
    "${UADE}/src/text_scope.c" "${UADE}/src/write_audio.c" "${UADE}/src/state_detection.c"
    "${UADE}/src/uade_logging.c"
    # The four files upstream's Makefile copies out of the frontend rather than sharing. Compiled
    # from where they live: a copy is a second thing to keep in step, and the copying was only ever
    # a make-rule convenience.
    "${UADE_COMMON}/uadeipc.c" "${UADE_COMMON}/uadeutils.c" "${UADE_COMMON}/unixatomic.c"
    "${UADE_COMMON}/ossupport.c")
target_include_directories(uadecore PRIVATE
    "${UADE}/src" "${UADE}/src/include" "${UADE}/src/frontends/include" "${UADE_COMMON}")
target_link_libraries(uadecore PRIVATE uade_zakalwe m)
target_compile_options(uadecore PRIVATE -w -fno-exceptions -fomit-frame-pointer)
set_target_properties(uadecore PROPERTIES
    PREFIX "lib" OUTPUT_NAME "uadecore" SUFFIX ".so"
    POSITION_INDEPENDENT_CODE ON)

# The engine links `uade_frontend` and compiles with this on; everything else in the tree is
# unaffected, and a build without UADE keeps every path through `openBackend` that exists today.
target_compile_definitions(uade_frontend INTERFACE PROTRACKTOR_WITH_UADE=1)
